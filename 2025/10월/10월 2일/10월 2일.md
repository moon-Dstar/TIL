# [TIL] Final Project 4일차

## JavaScript

### this
```
<script type="text/javascript">
        // 1. 일반함수 안에서 this는 window 객체이다.
        var data = 10;

        function func() {
            this.data = 20;
            data = 30;

            console.log('1. data: ' + data);
            console.log('2. this.data: ' + this.data);
            console.log('3. window.data: ' + window.data);
        }

        func();

	      // 2. 리터럴 객체, 생성자 함수, 프로토타입 안에서의 this는 해당 객체이다.
        var data = 10;

        function MyClass() {
            this.data = 0;
        }

        MyClass.prototype.method1 = function() {
            this.data = 20;
            data = 30;

            console.log('1. data: ' + data);
            console.log('2. this.data: ' + this.data);
            console.log('3. window.data: ' + window.data);
        }

        const m1 = new MyClass();
        m1.method1();

	      // 3. 이벤트 핸들러 함수 안에서 this는 이벤트 대상 객체이다.
        var data = 10;

        const btn = document.getElementById('btn');
        btn.onclick = function() {
            this.data = 20;
            data = 30;

            console.log('1. data: ' + data);
            console.log('2. this.data: ' + this.data);
            console.log('3. window.data: ' + window.data);
        }

	      var data = 'Global data';

        var obj1 = {data: 'obj1 data'};
        var obj2 = {data: 'obj2 data'};

        function func() {
            console.log(this.data);
        }
        // call, apply => this 결정
        func.call(obj1);
        func.apply(obj2);
</script>
```
